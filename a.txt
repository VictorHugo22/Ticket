const handleSelect = (ticket: Ticket) => {
  // Si hay comentario escrito en el ticket actual
  if (ticketDetailComentario.trim() !== "") {
    const confirmar = window.confirm(
      "Tienes comentarios sin guardar. Si cambias de ticket se perderán. ¿Deseas continuar?"
    );

    if (!confirmar) return; // cancelar el cambio de ticket si el usuario hace "Cancelar"
  }

  // Actualizar ticket seleccionado
  if (ticket.Estado?.id_estado === "2") {
    setSelectedTicket(ticket);
    setShowCommentsFor(ticket.id_ticket);
  } else {
    setSelectedTicket(ticket);
    setShowCommentsFor(null);
  }

  // Limpiar comentario solo si el usuario confirmó el cambio
  setTicketDetailComentario("");
};
